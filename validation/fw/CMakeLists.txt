cmake_minimum_required(VERSION 3.13)
# One RP2040 image for the concurrency study (see main.c).
#   PICO_SDK_PATH, FREERTOS_KERNEL_PATH: from the environment.
set(LIB "${CMAKE_CURRENT_LIST_DIR}/lib")
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
    set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT FREERTOS_KERNEL_PATH AND DEFINED ENV{FREERTOS_KERNEL_PATH})
    set(FREERTOS_KERNEL_PATH "$ENV{FREERTOS_KERNEL_PATH}")
endif()
set(PICO_PLATFORM rp2040)
set(PICO_BOARD pico)
add_compile_definitions(configUSE_TICKLESS_IDLE=0)
include(${LIB}/pico_sdk_import.cmake)
project(edg_pn_study C CXX ASM)
pico_sdk_init()
set(FREERTOS_CONFIG_FILE_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/config CACHE STRING "")
include(${LIB}/FreeRTOS_Kernel_import.cmake)

add_executable(study main.c ${LIB}/edg.c ${LIB}/edg_pn.c ${LIB}/edg_pn_choice.c)
set(DEFER_GROUP_INIT 1 CACHE STRING "1: initialize after scheduler starts; 0: reproduce original startup")
target_compile_definitions(study PRIVATE DEFER_GROUP_INIT=${DEFER_GROUP_INIT})
target_include_directories(study PRIVATE ${LIB} ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(study PRIVATE -Wall -Wextra)
target_link_libraries(study pico_stdlib FreeRTOS-Kernel-Heap4)
pico_enable_stdio_usb(study 0)
pico_enable_stdio_uart(study 1)
pico_add_extra_outputs(study)
